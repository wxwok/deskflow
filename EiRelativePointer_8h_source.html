15// Per-event motion beyond this is not a real pointer sample. One million
16// logical pixels is well past a frame-long flick on a large desktop, and
17// adding it to a cursor position cannot overflow int32_t.
18inline constexpr double kMaxRelativePointerDelta = 1000000.0;
19
20// Whole pixels to forward for one libei relative-pointer sample.
21// A dropped axis contributed nothing. reset is set when an axis accumulator
22// was non-finite or out of range and has been cleared.
23struct RelativePointerDelta
24{
25 std::int32_t x = 0;
26 std::int32_t y = 0;
27 bool droppedX = false;
28 bool droppedY = false;
29 bool reset = false;
30};

34// Apply one axis. Returns true when the sample was folded in (the pixel
35// count may still be zero when only a fraction of a pixel accumulated).
36inline bool accumulateRelativeAxis(double &buffer, double delta, std::int32_t &pixels, bool &reset)
37{
38 pixels = 0;
39
40 // A healthy accumulator only ever holds a fraction of a pixel. Anything
41 // else was stored by an earlier bad sample and must not be reused.
42 if (!std::isfinite(buffer) || std::fabs(buffer) > kMaxRelativePointerDelta) {
43 buffer = 0;
44 reset = true;
45 }
46
47 if (!std::isfinite(delta) || std::fabs(delta) > kMaxRelativePointerDelta) {
48 return false;
49 }
50
51 buffer += delta;
52
53 // static_cast<int32_t> is undefined when the truncated value does not fit.
54 // On x86, cvttsd2si returns INT_MIN (0x80000000) for NaN, infinity, and
55 // every out-of-range finite value. Subtracting that INT_MIN from the
56 // accumulator leaves NaN as NaN and pushes a huge finite value further
57 // out of range, so every later event on this axis also reports INT_MIN.
58 const double whole = std::trunc(buffer);
59 constexpr double kMinPixel = static_cast<double>(std::numeric_limits<std::int32_t>::min());
60 constexpr double kMaxPixel = static_cast<double>(std::numeric_limits<std::int32_t>::max());
61 if (!std::isfinite(whole) || whole < kMinPixel || whole > kMaxPixel) {
62 buffer = 0;
63 reset = true;
64 return false;
65 }
66
67 pixels = static_cast<std::int32_t>(whole);
68 buffer -= whole;
69 return true;
70}

72} // namespace detail

74// Fold one libei relative-pointer sample into the sub-pixel accumulators.
75//
76// ei_pointer.motion_relative carries IEEE-754 floats. Mutter sends a NaN when
77// input capture activates, and a finite sample can also fall outside the
78// int32 range. Either value used to stick in the accumulator (see
79// accumulateRelativeAxis). The server adds the resulting INT_MIN to the
80// cursor and switches computers near -2^31, which pins the pointer to the
81// client edge until the process restarts.
82inline RelativePointerDelta accumulateRelativePointer(double &bufferX, double &bufferY, double dx, double dy)
83{
84 RelativePointerDelta result;
85 bool resetX = false;
86 bool resetY = false;
87 result.droppedX = !detail::accumulateRelativeAxis(bufferX, dx, result.x, resetX);
88 result.droppedY = !detail::accumulateRelativeAxis(bufferY, dy, result.y, resetY);
89 result.reset = resetX || resetY;
90 return result;
91}
