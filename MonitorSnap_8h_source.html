66inline int64_t snapAbs(int32_t value)
67{
68 return value < 0 ? -static_cast<int64_t>(value) : static_cast<int64_t>(value);
69}

71inline bool snapOverlap(const SnapRect &a, const SnapRect &b)
72{
73 if (a.width <= 0 || a.height <= 0 || b.width <= 0 || b.height <= 0) {
74 return false;
75 }
76 return a.x < b.right() && b.x < a.right() && a.y < b.bottom() && b.y < a.bottom();
77}

80inline int32_t snapGap(int32_t a0, int32_t a1, int32_t b0, int32_t b1)
81{
82 if (a0 < b1 && b0 < a1) {
83 return 0;
84 }
85 if (a1 <= b0) {
86 return b0 - a1;
87 }
88 return a0 - b1;
89}

91inline bool
92groupOverlaps(const std::vector<SnapRect> &moving, const std::vector<SnapRect> &stationary, int32_t dx, int32_t dy)
93{
94 for (auto moved : moving) {
95 if (moved.width <= 0 || moved.height <= 0) {
96 continue;
97 }
98 moved.x += dx;
99 moved.y += dy;
100 for (const auto &still : stationary) {
101 if (snapOverlap(moved, still)) {
102 return true;
103 }
104 }
105 }
106 return false;
107}

109struct AxisCandidate

113 bool adjacency = false;

117inline bool betterAxis(const AxisCandidate &candidate, const AxisCandidate &current)
118{
119 const int64_t candidateDistance = snapAbs(candidate.delta);
120 const int64_t currentDistance = snapAbs(current.delta);
121 if (candidateDistance != currentDistance) {
122 return candidateDistance < currentDistance;
123 }
124 if (candidate.gap != current.gap) {
125 return candidate.gap < current.gap;
126 }
127 return candidate.adjacency && !current.adjacency;
128}
129
130inline void considerAxis(std::vector<AxisCandidate> &axis, AxisCandidate candidate, int32_t threshold)
131{
132 if (candidate.gap > threshold || snapAbs(candidate.delta) > threshold) {
133 return;
134 }
135 for (auto &existing : axis) {
136 if (existing.delta != candidate.delta) {
137 continue;
138 }
139 if (betterAxis(candidate, existing)) {
140 existing = candidate;
141 }
142 return;
143 }
144 axis.push_back(candidate);
145}

177inline bool
178separateGroup(const std::vector<SnapRect> &moving, const std::vector<SnapRect> &stationary, int32_t &dx, int32_t &dy)
179{
180 dx = 0;
181 dy = 0;
182 for (int iteration = 0; iteration < 8; ++iteration) {
183 if (!groupOverlaps(moving, stationary, dx, dy)) {
184 return true;
185 }
186
187 bool found = false;
188 int32_t bestDx = dx;
189 int32_t bestDy = dy;
190 int64_t bestCost = 0;
191 for (const auto &source : moving) {
192 SnapRect moved = source;
193 moved.x += dx;
194 moved.y += dy;
195 if (moved.width <= 0 || moved.height <= 0) {
196 continue;
197 }
198 for (const auto &still : stationary) {
199 if (!snapOverlap(moved, still)) {
200 continue;
201 }
202 const struct Push
203 {
204 int32_t x;
205 int32_t y;
206 } pushes[] = {
207 {still.x - moved.right(), 0},
208 {still.right() - moved.x, 0},
209 {0, still.y - moved.bottom()},
210 {0, still.bottom() - moved.y},
211 };
212 for (const auto &push : pushes) {
213 const int32_t nextDx = dx + push.x;
214 const int32_t nextDy = dy + push.y;
215 if (groupOverlaps(moving, stationary, nextDx, nextDy)) {
216 continue;
217 }
218 const int64_t cost = snapAbs(nextDx) + snapAbs(nextDy);
219 // Equal cost keeps the earlier push: left, then right, then up, then down.
220 if (!found || cost < bestCost) {
221 found = true;
222 bestCost = cost;
223 bestDx = nextDx;
224 bestDy = nextDy;
225 }
226 }
227 }
228 }
229 if (!found) {
230 return false;
231 }
232 dx = bestDx;
233 dy = bestDy;
234 }
235 return !groupOverlaps(moving, stationary, dx, dy);
236}

244inline std::vector<Interval> mergeIntervals(std::vector<Interval> intervals)
245{
246 if (intervals.empty()) {
247 return {};
248 }
249 std::sort(intervals.begin(), intervals.end(), [](const Interval &a, const Interval &b) { return a.begin < b.begin; });
250 std::vector<Interval> merged;
251 merged.push_back(intervals.front());
252 for (size_t index = 1; index < intervals.size(); ++index) {
253 if (intervals[index].begin <= merged.back().end) {
254 merged.back().end = (std::max)(merged.back().end, intervals[index].end);
255 } else {
256 merged.push_back(intervals[index]);
257 }
258 }
259 return merged;
260}

262inline std::vector<Interval> subtractIntervals(const std::vector<Interval> &base, std::vector<Interval> cuts)
263{
264 cuts = mergeIntervals(std::move(cuts));
265 std::vector<Interval> result;
266 for (const auto span : base) {
267 int32_t cursor = span.begin;
268 for (const auto cut : cuts) {
269 if (cut.end <= cursor) {
270 continue;
271 }
272 if (cut.begin >= span.end) {
273 break;
274 }
275 if (cut.begin > cursor) {
276 result.push_back({cursor, (std::min)(cut.begin, span.end)});
277 }
278 cursor = (std::max)(cursor, cut.end);
279 if (cursor >= span.end) {
280 break;
281 }
282 }
283 if (cursor < span.end) {
284 result.push_back({cursor, span.end});
285 }
286 }
287 return result;
288}

290inline void collectContacts(
291 std::vector<SwitchSegment> &segments, bool vertical, int32_t position, int32_t spanBegin, int32_t spanEnd,
292 std::vector<Interval> remote, std::vector<Interval> blocked
293)
294{
295 if (remote.empty()) {
296 return;
297 }
298 const std::vector<Interval> shown = subtractIntervals(mergeIntervals(std::move(remote)), std::move(blocked));
299 for (const auto interval : shown) {
300 if (interval.begin >= interval.end) {
301 continue;
302 }
303 const int32_t begin = (std::max)(interval.begin, spanBegin);
304 const int32_t end = (std::min)(interval.end, spanEnd);
305 if (begin < end) {
306 segments.push_back(SwitchSegment{vertical, position, begin, end});
307 }
308 }
309}

313inline MonitorSnapResult snapMonitorPlacement(
314 const std::vector<SnapRect> &moving, const std::vector<SnapRect> &stationary, int32_t threshold, bool enableSnap
315)
316{
317 using namespace detail;
318 if (threshold < 0) {
319 threshold = 0;
320 }
321
322 MonitorSnapResult result;
323 // Overlap is a stack, not a collision. Snapping off keeps the position that was asked for.
324 if (!enableSnap) {
325 return result;
326 }
327
328 std::vector<AxisCandidate> xCandidates;
329 std::vector<AxisCandidate> yCandidates;
330 for (const auto &moved : moving) {
331 if (moved.width <= 0 || moved.height <= 0) {
332 continue;
333 }
334 for (const auto &still : stationary) {
335 if (still.width <= 0 || still.height <= 0) {
336 continue;
337 }
338 const int32_t gapY = snapGap(moved.y, moved.bottom(), still.y, still.bottom());
339 const int32_t gapX = snapGap(moved.x, moved.right(), still.x, still.right());
340 considerAxis(
341 xCandidates,
342 {still.right() - moved.x, gapY, true,
343 guideOn(true, still.right(), moved.y, moved.bottom(), still.y, still.bottom())},
344 threshold
345 );
346 considerAxis(
347 xCandidates,
348 {still.x - moved.right(), gapY, true,
349 guideOn(true, still.x, moved.y, moved.bottom(), still.y, still.bottom())},
350 threshold
351 );
352 considerAxis(
353 xCandidates,
354 {still.x - moved.x, gapY, false, guideOn(true, still.x, moved.y, moved.bottom(), still.y, still.bottom())},
355 threshold
356 );
357 considerAxis(
358 xCandidates,
359 {still.right() - moved.right(), gapY, false,
360 guideOn(true, still.right(), moved.y, moved.bottom(), still.y, still.bottom())},
361 threshold
362 );
363 considerAxis(
364 yCandidates,
365 {still.bottom() - moved.y, gapX, true,
366 guideOn(false, still.bottom(), moved.x, moved.right(), still.x, still.right())},
367 threshold
368 );
369 considerAxis(
370 yCandidates,
371 {still.y - moved.bottom(), gapX, true,
372 guideOn(false, still.y, moved.x, moved.right(), still.x, still.right())},
373 threshold
374 );
375 considerAxis(
376 yCandidates,
377 {still.y - moved.y, gapX, false, guideOn(false, still.y, moved.x, moved.right(), still.x, still.right())},
378 threshold
379 );
380 considerAxis(
381 yCandidates,
382 {still.bottom() - moved.bottom(), gapX, false,
383 guideOn(false, still.bottom(), moved.x, moved.right(), still.x, still.right())},
384 threshold
385 );
386 }
387 }
388
389 const std::vector<AxisOption> xOptions = axisOptions(xCandidates);
390 const std::vector<AxisOption> yOptions = axisOptions(yCandidates);
391 bool chosen = false;
392 int chosenSnaps = -1;
393 int64_t chosenDistance = 0;
394 int chosenAdjacency = -1;
395 int64_t chosenAbsX = 0;
396 for (const auto &xOption : xOptions) {
397 for (const auto &yOption : yOptions) {
398 const int snaps = (xOption.snap ? 1 : 0) + (yOption.snap ? 1 : 0);
399 const int64_t distance = snapAbs(xOption.delta) + snapAbs(yOption.delta);
400 const int adjacency = (xOption.adjacency ? 1 : 0) + (yOption.adjacency ? 1 : 0);
401 const int64_t absX = snapAbs(xOption.delta);
402 const bool better = !chosen || snaps > chosenSnaps ||
403 (snaps == chosenSnaps &&
404 (distance < chosenDistance ||
405 (distance == chosenDistance &&
406 (adjacency > chosenAdjacency || (adjacency == chosenAdjacency && absX < chosenAbsX)))));
407 if (!better) {
408 continue;
409 }
410 chosen = true;
411 chosenSnaps = snaps;
412 chosenDistance = distance;
413 chosenAdjacency = adjacency;
414 chosenAbsX = absX;
415 result.dx = xOption.delta;
416 result.dy = yOption.delta;
417 result.snappedX = xOption.snap;
418 result.snappedY = yOption.snap;
419 result.guides.clear();
420 if (xOption.snap) {
421 result.guides.push_back(xOption.guide);
422 }
423 if (yOption.snap) {
424 result.guides.push_back(yOption.guide);
425 }
426 }
427 }
428
429 return result;
430}

432inline std::vector<SwitchSegment> switchEdgeSegments(const std::vector<SnapRect> &monitors)
433{
434 using namespace detail;
435 std::vector<SwitchSegment> segments;
436 for (const auto &monitor : monitors) {
437 if (monitor.width <= 0 || monitor.height <= 0) {
438 continue;
439 }
440
441 std::vector<Interval> rightRemote;
442 std::vector<Interval> rightBlocked;
443 std::vector<Interval> bottomRemote;
444 std::vector<Interval> bottomBlocked;
445 for (const auto &other : monitors) {
446 if (&other == &monitor || other.width <= 0 || other.height <= 0) {
447 continue;
448 }
449 if (other.x == monitor.right()) {
450 const int32_t begin = (std::max)(monitor.y, other.y);
451 const int32_t end = (std::min)(monitor.bottom(), other.bottom());
452 if (begin < end) {
453 (other.group == monitor.group ? rightBlocked : rightRemote).push_back({begin, end});
454 }
455 }
456 if (other.y == monitor.bottom()) {
457 const int32_t begin = (std::max)(monitor.x, other.x);
458 const int32_t end = (std::min)(monitor.right(), other.right());
459 if (begin < end) {
460 (other.group == monitor.group ? bottomBlocked : bottomRemote).push_back({begin, end});
461 }
462 }
463 }
464 collectContacts(
465 segments, true, monitor.right(), monitor.y, monitor.bottom(), std::move(rightRemote), std::move(rightBlocked)
466 );
467 collectContacts(
468 segments, false, monitor.bottom(), monitor.x, monitor.right(), std::move(bottomRemote), std::move(bottomBlocked)
469 );
470 }
471 return segments;
472}
